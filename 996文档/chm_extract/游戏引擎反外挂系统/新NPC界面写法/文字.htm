-- -- simplenum  简化数值 1 （超万显示万为单位 超亿显示亿为单位）
-- -- scrollWidth   滚动区域宽度
-- -- scrollWay     滚动方式 -- 0 从右到左 1 从下到上
-- -- scrollTime    滚动时间(秒)
-- -- scrollHeight  滚动区域高度
