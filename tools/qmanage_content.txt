[@Startup]
; 点卡系统初始化 - 使用A500存储点卡时间(秒)

[@Login]
#IF
#act
ChangeBagCount = 126

;================================================
; 点卡系统 - 登录时检测
;================================================
; 启动51号定时器（每秒检测是否非法离开600地图）
SETONTIMER 51 1

; 7级以下免费玩，不检查点卡
#IF
SMALL <$LEVEL> 7
#ACT
SENDMSG 6 欢迎！7级之前免费畅玩！
BREAK

; 7级或以上，检查点卡
; A500存储点卡时间(秒)
#IF
EQUAL A500 0
#ACT
MAPMOVE 600 20 20
SENDMSG 5 您的游戏时间已用完，请充值后使用点卡续费！
BREAK

; 有时间，显示剩余
#IF
LARGE A500 0
#ACT
SENDMSG 6 您好！当前剩余游戏时间：<$STR(A500)>秒
SETONTIMER 50 3600
; 欠费区回来且有点卡时间，自动传回土城
#IF
CHECKMAPNAME 600
#ACT
MAPMOVE 3 330 330
SENDMSG 6 已自动传送回安全区！
BREAK

;================================================
#IF
; 在副本内被怪物杀死
CHECKMAPNAME BRTL
#ACT
HumanDropUseItem -1 S99
HumanDropUseItem -1 S98
MOV A200 0
MOV A201 0
MAPMOVE 3 330 330
BREAK

#IF
#ACT
BREAK

;================================================
; 死亡前触发 - 复活符（副本内使用一次复活）
; A202=1 表示本局拥有一次复活机会
;================================================
[@NextDie]
#IF
CHECKMAPNAME BRTL
EQUAL A202 1
#ACT
MOV A202 0
ChangeModeEx 23 1 1
SENDMSG 5 【复活符】在原地复活了！复活机会已用尽！
BREAK

#IF
#ACT
BREAK

;================================================
#IF
#ACT
MOV G302 +1

; ===== 第1阶段：G302=1(30秒) -> 正式开始广播+刷基础怪 =====
#IF
EQUAL G302 1
#ACT
MonGen 稻草人 20 0
MonGen 野鸡 30 0
MonGen 鹿 20 0
MonGen 半兽人 15 0
BREAK

; ===== 第2阶段：G302=20(10分钟) -> 骷髅僵尸 =====
#IF
EQUAL G302 20
#ACT
MonGen 骷髅精灵 20 0
MonGen 骷髅武士 20 0
MonGen 腐尸 15 0
MonGen 多钩猫 10 0
BREAK

; ===== 第3阶段：G302=40(20分钟) -> 精英怪 =====
#IF
EQUAL G302 40
#ACT
MonGen 祖玛卫士 15 0
MonGen 祖玛教主 5 0
MonGen 赤月恶魔 5 0
BREAK

; ===== 第4阶段：G302=50(25分钟) -> 毒圈预警 =====
#IF
EQUAL G302 50
#ACT
SETONTIMER 61 10
BREAK

; ===== 第5阶段：G302=60(30分钟) -> 副本结束 =====
#IF
EQUAL G302 60
#ACT
SETOFFTIMER 60
SETOFFTIMER 61
BREAK

; ===== 每10轮(5分钟)刷一次空投宝箱 =====
#IF
EQUAL D99 0
#ACT
BREAK

#IF
#ACT

;----------------------------------------------------
[@OnTimer51]
; 每秒检测一次，防止在600地图使用回城卷飞出
#IF
ISONMAP 600
#ACT
; 在600地图，延迟1秒后继续检测
DELAYGOTO 1000 @OnTimer51
#ELSEACT
; 不在600地图了，可能是用了回城卷，检查是否有游戏时间
GOTO @检查点卡状态

[@检查点卡状态]
; 检查是否有游戏时间(A500)
#IF
EQUAL A500 0
#ACT
; 没有游戏时间，强制传送回600
MAPMOVE 600 20 20
SENDMSG 5 欠费地图无法使用回城卷！请充值后使用点卡续费！
#ELSEACT
; 有游戏时间，允许离开，停止定时器
SETONTIMER 51 0
SENDMSG 6 您已离开欠费地图，祝您游戏愉快！

;================================================
; 50号定时器 - 每小时扣减点卡时间
;================================================
[@OnTimer50]
; 扣减1小时(3600秒)
#IF
LARGE A500 3600
#ACT
DEC A500 3600
SENDMSG 6 点卡时间已扣除1小时，剩余时间：<$STR(A500)>秒
#ELSEACT
; 剩余时间不足1小时，直接清零
MOV A500 0
SENDMSG 5 您的游戏时间已用完，请充值后使用点卡续费！
; 传送到欠费区
MAPMOVE 600 20 20
