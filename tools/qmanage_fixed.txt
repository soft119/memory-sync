[@Startup]
; 点卡系统初始化

[@Login]
#IF
#act
ChangeBagCount = 126

;================================================
; 点卡系统 - 登录时检测
;================================================
; 7级以下免费玩，不检查点卡
#IF
SMALL <$LEVEL> 7
#ACT
SENDMSG 6 欢迎！7级之前免费畅玩！
BREAK

; 7级或以上，检查是否在欠费区
#IF
CHECKMAPNAME 600
#ACT
; 在欠费区，检查是否有时间
#IF
LARGE A500 1
#ACT
; 有时间，传回土城
MAPMOVE 3 330 330
SENDMSG 6 已自动传送回安全区！
BREAK
#ELSEACT
; 没时间，留在欠费区
SENDMSG 5 您的游戏时间已用完，请充值后使用点卡续费！
BREAK

; 不在欠费区，检查是否有游戏时间
#IF
SMALL A500 1
#ACT
; 没时间，传送到欠费区
MAPMOVE 600 20 20
SENDMSG 5 您的游戏时间已用完，请充值后使用点卡续费！
BREAK

; 有时间，正常登录
#IF
#ACT
SENDMSG 6 您好！当前剩余游戏时间：<$STR(A500)>秒
; 启动50号定时器（每小时扣减）
SETONTIMER 50 3600

;================================================
#IF
; 在副本内被怪物杀死
CHECKMAPNAME BRTL
#ACT
HumanDropUseItem -1 S99
HumanDropUseItem -1 S98
MOV A200 0
MOV A201 0
MAPMOVE 3 330 330
BREAK

#IF
#ACT
BREAK

;================================================
; 死亡前触发 - 复活符（副本内使用一次复活）
; A202=1 表示本局拥有一次复活机会
;================================================
[@NextDie]
#IF
CHECKMAPNAME BRTL
EQUAL A202 1
#ACT
MOV A202 0
ChangeModeEx 23 1 1
SENDMSG 5 【复活符】在原地复活了！复活机会已用尽！
BREAK

#IF
#ACT
BREAK

;================================================
#IF
#ACT
MOV G302 +1

; ===== 第1阶段：G302=1(30秒) -> 正式开始广播+刷基础怪 =====
#IF
EQUAL G302 1
#ACT
MonGen 稻草人 20 0
MonGen 野鸡 30 0
MonGen 鹿 20 0
MonGen 半兽人 15 0
BREAK

; ===== 第2阶段：G302=20(10分钟) -> 骷髅僵尸 =====
#IF
EQUAL G302 20
#ACT
MonGen 骷髅精灵 20 0
MonGen 骷髅武士 20 0
MonGen 腐尸 15 0
MonGen 多钩猫 10 0
BREAK

; ===== 第3阶段：G302=40(20分钟) -> 精英怪 =====
#IF
EQUAL G302 40
#ACT
MonGen 祖玛卫士 15 0
MonGen 祖玛教主 5 0
MonGen 赤月恶魔 5 0
BREAK

; ===== 第4阶段：G302=50(25分钟) -> 毒圈预警 =====
#IF
EQUAL G302 50
#ACT
SETONTIMER 61 10
BREAK

; ===== 第5阶段：G302=60(30分钟) -> 副本结束 =====
#IF
EQUAL G302 60
#ACT
SETOFFTIMER 60
SETOFFTIMER 61
BREAK

; ===== 每10轮(5分钟)刷一次空投宝箱 =====
#IF
EQUAL D99 0
#ACT
BREAK

#IF
#ACT

;================================================
; 50号定时器 - 每小时扣减点卡时间
;================================================
[@OnTimer50]
; 扣减1小时(3600秒)
#IF
LARGE A500 3600
#ACT
DEC A500 3600
SENDMSG 6 点卡时间已扣除1小时，剩余时间：<$STR(A500)>秒
#ELSEACT
; 剩余时间不足1小时，清零并传送
MOV A500 0
SENDMSG 5 您的游戏时间已用完，请充值后使用点卡续费！
MAPMOVE 600 20 20
