#IF
#ACT

;================================================
#IF
CHECKLEVELEX = 1
#ACT
VAR Integer HUMAN 新人礼包

#IF
CHECKLEVELEX = 1
CHECKVAR HUMAN 新人礼包 < 1
#ACT
GIVE 木剑 1
GIVE 蜡烛 1
CALCVAR HUMAN 新人礼包 = 1
SAVEVAR HUMAN 新人礼包

#IF
CHECKLEVELEX = 1
GENDER MAN
CHECKVAR HUMAN 新人礼包 = 1
#ACT
GIVE 布衣(男) 1
SENDMSG 6 欢迎新人！系统赠送您：木剑、布衣(男)、蜡烛！
SENDMSG 0 玩家「<$USERNAME>」进入了游戏，欢迎新朋友！

#IF
CHECKLEVELEX = 1
GENDER WOMAN
CHECKVAR HUMAN 新人礼包 = 1
#ACT
GIVE 布衣(女) 1
SENDMSG 6 欢迎新人！系统赠送您：木剑、布衣(女)、蜡烛！
SENDMSG 0 玩家「<$USERNAME>」进入了游戏，欢迎新朋友！

;================================================
; 点卡系统 - 登录检测
; 22级以下新人不需要点卡
;================================================
#IF
CHECKLEVELEX < 22
#ACT
SENDMSG 6 欢迎新人！22级以下免费游戏，祝您游戏愉快！
BREAK

; 22级及以上需要检测点卡
#IF
#ACT
VAR Integer HUMAN 点卡时间
LOADVAR HUMAN 点卡时间 ..\QuestDiary\系统功能\点卡时间.txt

#IF
CHECKVAR HUMAN 点卡时间 < 1
#ACT
MAPMOVE 600 20 20
SENDMSG 5 您已达到22级，需要点卡才能继续游戏！
SENDMSG 5 请使用7天点卡续费！
BREAK

; 有游戏时间，计算并显示剩余时间
#IF
#ACT
DIV N2 <$HUMAN(点卡时间)> 86400
MUL N3 <$STR(N2)> 86400
MOV N4 <$HUMAN(点卡时间)>
DEC N4 <$STR(N3)>
DIV N5 <$STR(N4)> 3600
MUL N6 <$STR(N5)> 3600
MOV N7 <$STR(N4)>
DEC N7 <$STR(N6)>
DIV N8 <$STR(N7)> 60
SENDMSG 6 您好！当前剩余游戏时间：<$STR(N2)>天<$STR(N5)>小时<$STR(N8)>分钟
SETONTIMER 50 60

;================================================
[@OnTimer50]
#IF
#ACT
VAR Integer HUMAN 点卡时间
LOADVAR HUMAN 点卡时间 ..\QuestDiary\系统功能\点卡时间.txt

#IF
CHECKVAR HUMAN 点卡时间 > 60
#ACT
CALCVAR HUMAN 点卡时间 - 60
SAVEVAR HUMAN 点卡时间 ..\QuestDiary\系统功能\点卡时间.txt
SETONTIMER 50 60

#ELSEACT
CALCVAR HUMAN 点卡时间 = 0
SAVEVAR HUMAN 点卡时间 ..\QuestDiary\系统功能\点卡时间.txt
